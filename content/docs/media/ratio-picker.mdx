```bash
npx shadcn@latest add https://motif.sweetretry.com/r/ratio-picker.json
```

## Usage

```tsx
import { RatioPicker } from "@/components/ui/motif/ratio-picker";

export function Frame() {
  const [ratio, setRatio] = useState("16:9");

  return (
    <div className="space-y-2">
      <div className="font-medium text-muted-foreground text-xs">
        Aspect ratio
      </div>
      <RatioPicker
        aria-label="Aspect ratio"
        onChange={setRatio}
        value={ratio}
      />
    </div>
  );
}
```

- Renders no label; pass `aria-label` alongside your own heading.
- Pass custom `options` when needed (`"adaptive"` and `"auto"` render a scan glyph); `RatioIcon` is exported standalone for triggers or badges.
- `showReferenceBox` draws a dashed square frame around the selected glyph.

```tsx
<RatioPicker
  aria-label="Canvas"
  options={["1:1", "16:9", "9:16", "adaptive"]}
  value={ratio}
  onChange={setRatio}
/>
```

## Accessibility

- Renders `role="group"` named by `aria-label` with `<button aria-pressed>` cells; glyphs are `aria-hidden`.
