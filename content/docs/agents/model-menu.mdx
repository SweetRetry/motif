```bash
npx shadcn@latest add https://motif.sweetretry.com/r/model-menu.json
```

## Usage

```tsx
import {
  ModelMenu,
  ModelMenuContent,
  ModelMenuEffort,
  ModelMenuModels,
  ModelMenuTrigger,
} from "@/components/ui/motif/model-menu";

export function Composer() {
  const [model, setModel] = useState("deepseek-v4-flash");
  const [effort, setEffort] = useState<ReasoningEffort | null>(null);

  return (
    <ModelMenu
      effort={effort}
      models={MODELS}
      onEffortChange={setEffort}
      onValueChange={setModel}
      value={model}
    >
      <ModelMenuTrigger />
      <ModelMenuContent>
        <ModelMenuModels />
        <ModelMenuEffort />
      </ModelMenuContent>
    </ModelMenu>
  );
}
```

Each model declares its supported `thinking` options and default:

```tsx
{
  id: "claude-sonnet-5-5",
  name: "Claude Sonnet 5.5",
  provider: "anthropic",
  family: "anthropic",
  thinking: { options: ["low", "medium", "high", "xhigh", "max"], default: "high" },
}
```

## Parts

| Part               | Role                                                                   |
| ------------------ | ---------------------------------------------------------------------- |
| `ModelMenu`        | Root state provider for `value`, `effort`, `models`, and `family` lock |
| `ModelMenuTrigger` | Displays provider mark, model name, effort, and chevron (`children` overrides) |
| `ModelMenuContent` | Upward-opening menu panel                                              |
| `ModelMenuModels`  | Model radio list; accepts `label` and `lockedLabel` headings           |
| `ModelMenuEffort`  | Submenu row for reasoning effort; omitted when the model has ≤1 option |

## Behavior & Accessibility

- Unlike [Model Select](/docs/media/model-select) (searchable catalog), `ModelMenu` pairs chat models with per-model reasoning steps.
- The user's `effort` selection is preserved across model switches; use `effortOf(model, effort)` to resolve the effective step (falling back to the model's default when unsupported).
- Passing `family` disables models from other protocol families under `lockedLabel`.

<ModelMenuLockedDemo />

- Built on Radix radio groups for arrow-key navigation; trigger `aria-label` announces model and effort together while provider marks are `aria-hidden`.
